\section*{附录 C：控制层的补充推导}

\subsection*{C.1 扰动项的显式性与 $\alpha$ 条件的作用}

$\boldsymbol\tau=\hat M\ddot{\boldsymbol q}_{\mathrm{ref}}+\hat C\dot{\boldsymbol q}+\hat g$ 代入真实动力学 $M\ddot{\boldsymbol q}+C\dot{\boldsymbol q}+\boldsymbol g=\boldsymbol\tau+\delta\boldsymbol\tau+\boldsymbol\tau_{\mathrm{ext}}$ 并解出 $\ddot{\boldsymbol q}$：
\begin{equation*}
\ddot{\boldsymbol q}=M^{-1}\hat M\ddot{\boldsymbol q}_{\mathrm{ref}}+M^{-1}\bigl(\Delta C\dot{\boldsymbol q}+\Delta\boldsymbol g+\delta\boldsymbol\tau+\boldsymbol\tau_{\mathrm{ext}}\bigr),
\qquad M^{-1}\hat M=I+M^{-1}\Delta M ,
\end{equation*}
即 \eqref{eq:5.1}。$\ddot{\boldsymbol q}_{\mathrm{ref}}$ 由 \eqref{eq:5.2} 完全由当前可测量 $(\boldsymbol q,\dot{\boldsymbol q},\hat{\underline x}_d,\boldsymbol\xi_d,\dot{\boldsymbol\xi}_d)$ 给出、不依赖 $\ddot{\boldsymbol q}$，故 \eqref{eq:5.1} 对 $\ddot{\boldsymbol q}$ 是显式赋值。

$\alpha<1/\mathrm{cond}_2(K_d)$（\eqref{eq:5.1f}）并非适定性条件，而是保证有效阻尼正定：由 $|e_\xi^\top\Theta(-K_de_\xi)|\le\alpha\lambda_{\max}(K_d)\|e_\xi\|^2$ 得 $\lambda_{\mathrm{eff}}=\lambda_{\min}(K_d)-\alpha\lambda_{\max}(K_d)$，其正性即 \eqref{eq:5.1f}。$\alpha$ 过大时闭环仍定义良好，但 Lyapunov 证书失效。

\subsection*{C.2 定理 3(b)：$A$ 的奇异值下界}

C.6 第 4 步的 LaSalle 论证需要 $A$ 在工作域内一致可逆。定量化如下。

由 \eqref{eq:4.5}，$A=\begin{bmatrix}A_{11}&0\\ -[\mathcal T]_\times& I_3\end{bmatrix}$，$A_{11}=-\tfrac12(\tilde\eta I+[\mathcal O]_\times)$。其逆为
\begin{equation*}
\begin{gathered}
A^{-1}=\begin{bmatrix}A_{11}^{-1}&0\\ [\mathcal T]_\times A_{11}^{-1}& I_3\end{bmatrix},\\[2pt]
A_{11}^{-1}=-2\,\frac{\tilde\eta^2I+\tilde\eta[\mathcal O]_\times^{\top}+\mathcal O\mathcal O^\top}{\tilde\eta(\tilde\eta^2+\|\mathcal O\|^2)}
=-\frac2{\tilde\eta}\bigl(\tilde\eta^2I-\tilde\eta[\mathcal O]_\times+\mathcal O\mathcal O^\top\bigr),
\end{gathered}
\end{equation*}
用到 $\tilde\eta^2+\|\mathcal O\|^2=1$。记 $B\triangleq\tilde\eta I+[\mathcal O]_\times$（故 $A_{11}=-\tfrac12B$），则
\begin{equation*}
B^\top B=(\tilde\eta I-[\mathcal O]_\times)(\tilde\eta I+[\mathcal O]_\times)=\tilde\eta^2I-[\mathcal O]_\times^2=(\tilde\eta^2+\|\mathcal O\|^2)I-\mathcal O\mathcal O^\top=I-\mathcal O\mathcal O^\top ,
\end{equation*}
其特征值为 $\{\tilde\eta^2,1,1\}$，故 $B$ 的奇异值为 $\{\tilde\eta,1,1\}$，得
\begin{equation*}
\|A_{11}\|_2=\tfrac12\sigma_{\max}(B)=\tfrac12 ,
\quad
\|A_{11}^{-1}\|_2=\frac2{\sigma_{\min}(B)}=\frac2{\tilde\eta} ,
\end{equation*}
由分块下三角结构 $\|A^{-1}\|_2\le(1+\|\mathcal T\|)\allowbreak\|A_{11}^{-1}\|_2+1$，
\begin{equation*}
\sigma_{\min}(A)=\frac1{\|A^{-1}\|_2}\;\ge\;\Bigl[\frac{2(1+\|\mathcal T\|)}{\tilde\eta}+1\Bigr]^{-1} .
\end{equation*}
在 $\Omega_c$ 内 $\tilde\eta\ge\eta_0$（\eqref{eq:5.5c}）且 $\|\mathcal T\|\le\sqrt{2c/\lambda_{\min}(K_{p,T})}$，故 $\sigma_{\min}(A)\ge c_A(\eta_0,c)>0$ 一致成立。

\subsection*{C.3 定理 3(c-2) 拆分条件的细节}

拆分依赖两条恒等式：(i) $(A^\top e_z)_\omega$ 中 $[\mathcal T]_\times\mathcal T=0$；(ii) $\dot{\mathcal T}=-[\mathcal T]_\times\tilde\omega+\tilde v$ 中 $\mathcal T^\top(\mathcal T\times\tilde\omega)=0$。二者为代数恒等式，不依赖工作域，故 (c-2) 与 (c-1) 同为全局结论；$K_d$ 非块对角时 $-e_\xi^\top K_de_\xi$ 含 $\tilde\omega^\top K_{\omega v}\tilde v$ 交叉项，两分量能量不再分离，退回 (c-1)。

$K_{p,T}$ 各向同性的必要性：取一般对称正定 $K_{p,T}$ 时两处恒零均失效：(i) $(A^\top K_pe_z)_\omega=A_{11}^\top K_{p,O}\mathcal O+[\mathcal T]_\times K_{p,T}\mathcal T$，而 $[\mathcal T]_\times K_{p,T}\mathcal T\ne0$ 除非 $\mathcal T$ 为 $K_{p,T}$ 的特征向量；(ii) $\tfrac12\mathcal T^\top K_{p,T}\mathcal T$ 的导数含耦合项 $-\mathcal T^\top K_{p,T}[\mathcal T]_\times\tilde\omega$，仅当 $K_{p,T}=k_{p,T}I_3$ 时为零。残留耦合使 $V_\omega,V_v$ 不再各自闭合。

\subsection*{C.4 strictification 与局部 ISS（未来工作）}

定理 3(d) 缺口的根源是 $\dot V$ 中无 $-\|e_z\|^2$ 项。标准补救为 strictification：取 $W=V+\epsilon e_z^\top K_pA(\tilde x)e_\xi$，$\epsilon$ 足够小时 $W$ 与 $V$ 在 $\Omega_c$ 内等价，求导新增 $-\epsilon e_z^\top K_pAA^\top K_pe_z$ 提供 $e_z$ 方向负定；若剩余交叉项（含 $\dot A$ 项）可被两个负定项吸收，则 $\dot W\le-c_1\|(e_z,e_\xi)\|^2+c_2\|d\|\cdot\|(e_z,e_\xi)\|$，即局部 ISS-Lyapunov 函数。该验证（$\dot A$ 项的一致界与 $\epsilon$ 可行区间非空性）本文未完成，列为 §7 局限。

\subsection*{C.6 定理 3(b) 的完整证明}

\begin{enumerate}[leftmargin=2em]
  \item \textbf{交叉项精确相消}：沿 \eqref{eq:5.5}（$d\equiv0$），
  \begin{equation*}
  \dot V=e_\xi^\top\dot e_\xi+e_z^\top K_p\dot e_z
  =e_\xi^\top\bigl(-K_de_\xi-A^\top K_pe_z\bigr)+e_z^\top K_pAe_\xi
  =-e_\xi^\top K_de_\xi ,
  \end{equation*}
  因 $e_z^\top K_pAe_\xi=(A^\top K_pe_z)^\top e_\xi$（只用 $K_p$ 对称），$\dot V\le0$，$\Omega_c$ 正向不变；$V\le c$ 给出 $\|e_\xi\|\le\sqrt{2c}$、$\|e_z\|\le\sqrt{2c/\lambda_{\min}(K_p)}$，故 $\Omega_c$ 紧。

  \item \textbf{工作域保号 \eqref{eq:5.5c}}：$V\le c$ 蕴含 $\tfrac12\mathcal O^\top K_{p,O}\mathcal O\le c$，即 $\|\mathcal O\|^2\le2c/\lambda_{\min}(K_{p,O})<1$（由 $c<c^*$）。由 $\mathcal O=-\mathrm{Im}\,\tilde r$ 与 $\|\tilde r\|=1$ 得 $\tilde\eta^2+\|\mathcal O\|^2=1$，故 $|\tilde\eta|\ge\eta_0>0$；$\tilde\eta(t)$ 连续且恒非零，由 $\tilde\eta(0)>0$ 得 \eqref{eq:5.5c}。

  \item \textbf{$A$ 的行列式}：$A$ 块下三角（\eqref{eq:4.5}），故
  \begin{equation*}
  \det A=\det\bigl(-\tfrac12(\tilde\eta I_3+[\mathcal O]_\times)\bigr)\cdot\det I_3
  =\bigl(-\tfrac12\bigr)^3\tilde\eta\bigl(\tilde\eta^2+\|\mathcal O\|^2\bigr)=-\tfrac18\tilde\eta ,
  \end{equation*}
  用到 $\det(aI_3+[b]_\times)=a(a^2+\|b\|^2)$ 与 $\tilde\eta^2+\|\mathcal O\|^2=1$。定量奇异值下界 $\sigma_{\min}(A)\ge\bigl[2(1+\|\mathcal T\|)/\tilde\eta+1\bigr]^{-1}$ 见附录 C.2。

  \item \textbf{LaSalle}：$\Omega_c$ 紧且不变，$E\triangleq\{\dot V=0\}\cap\Omega_c=\{e_\xi=0\}\cap\Omega_c$。若轨迹全程留在 $E$ 内：$e_\xi\equiv0\Rightarrow\dot e_\xi\equiv0\Rightarrow A^\top K_pe_z\equiv0$；由第 3 步 $A$ 可逆、$K_p\succ0$ 得 $e_z\equiv0$。故 $E$ 内最大不变集为 $\{(0,0)\}$，由 LaSalle 不变集定理（\cite{Kha02} Thm 4.4）得 (iii)。

  \item \textbf{局部指数稳定}：在 $(0,0)$ 处 $\tilde x\to1$，$A\to A_0=\mathrm{diag}(-\tfrac12I_3,I_3)$，\eqref{eq:5.5} 的雅可比为
  \begin{equation*}
  F=\begin{bmatrix}0_6 & A_0\\ -A_0^\top K_p & -K_d\end{bmatrix}.
  \end{equation*}
  对该 LTI 系统同一 $V$ 给出 $\dot V\le0$，$A_0$ 可逆，重复第 4 步得 $F$ Hurwitz；由 Lyapunov 线化定理（\cite{Kha02} Thm 4.7）得局部指数稳定。块对角 $K_d,K_p$ 下 $F$ 逐分量解耦，两通道极点即 $\{-a,-b\}$。
\end{enumerate}

\subsection*{C.7 定理 3(c) 的完整证明}

\textbf{第一步（$\dot V$ 精确式）}：含扰时交叉项相消与 $d$ 无关，故
\begin{equation*}
\dot V=-e_\xi^\top K_de_\xi+e_\xi^\top d
\end{equation*}
精确成立。

\textbf{第二步（(c-1) 判据的等价性）}：性能目标即 $-e_\xi^\top K_de_\xi+e_\xi^\top d+\tfrac1{2\kappa}\|e_\xi\|^2-\tfrac{\gamma_a^2}2\|d\|^2\le0$，等价于二次型不等式
\begin{equation*}
\begin{bmatrix}e_\xi\\ d\end{bmatrix}^{\!\top}\!M\!
\begin{bmatrix}e_\xi\\ d\end{bmatrix}\ge0\quad\forall(e_\xi,d)\in\mathbb R^{12},
\end{equation*}
即 $M\succeq0$。右下块 $\tfrac{\gamma_a^2}2I\succ0$，取 Schur 补得 $K_d-\tfrac1{2\kappa}I-\tfrac1{2\gamma_a^2}I\succeq0$，即 \eqref{eq:5.6a}。

\textbf{第四步（积分）}：由 $M\succeq0$ 知 $V$ 一致有界、解在 $[0,\infty)$ 上存在，故可在 $[0,T]$ 上积分 $\dot V\le-\tfrac1{2\kappa}\|e_\xi\|^2+\tfrac{\gamma_a^2}2\|d\|^2$，弃去 $V(T)\ge0$，令 $T\to\infty$（单调收敛）即得 \eqref{eq:5.6}。

\textbf{第五步（(c-2) 分量解耦）}：由 \eqref{eq:4.5}，$(A^\top K_pe_z)_\omega=A_{11}^\top K_{p,O}\mathcal O+k_{p,T}[\mathcal T]_\times\mathcal T=A_{11}^\top K_{p,O}\mathcal O$（$\mathcal T\times\mathcal T=0$）、$(A^\top K_pe_z)_v=k_{p,T}\mathcal T$，且 $\dot{\mathcal T}=-[\mathcal T]_\times\tilde\omega+\tilde v$ 中 $\mathcal T^\top(\mathcal T\times\tilde\omega)=0$，故两分量内位姿交叉项分别精确相消，
\begin{equation*}
\dot V_\omega=-\tilde\omega^\top K_\omega\tilde\omega+\tilde\omega^\top d_\omega,
\qquad
\dot V_v=-\tilde v^\top K_v\tilde v+\tilde v^\top d_v ,
\end{equation*}
对每个分量重复第二至第四步（$I_6\to I_3$）即得 \eqref{eq:5.6b}$\Rightarrow$\eqref{eq:5.6p}。

\subsection*{C.8 定理 3(d) 的完整证明}

\begin{enumerate}[leftmargin=2em]
  \item \textbf{乘法项展开}：$\dot V=-e_\xi^\top K_de_\xi+e_\xi^\top d$（C.7 第一步）。代入 \eqref{eq:5.1d} 与 $u_{\mathrm{fb}}=-K_de_\xi-A^\top K_pe_z$：
  \begin{equation*}
  \dot V=-e_\xi^\top K_de_\xi\;\underbrace{-\,e_\xi^\top\Theta K_de_\xi}_{\text{与阻尼同类}}\;\underbrace{-\,e_\xi^\top\Theta A^\top K_pe_z}_{\text{与扰动同类}}\;+\;e_\xi^\top d_{\mathrm{ex}} .
  \end{equation*}

  \item \textbf{分别回收}：$|e_\xi^\top\Theta K_de_\xi|\le\alpha\lambda_{\max}(K_d)\|e_\xi\|^2$ 并入阻尼，由 \eqref{eq:5.1f} 得 \eqref{eq:5.7a} 的 $\lambda_{\mathrm{eff}}>0$；$|e_\xi^\top\Theta A^\top K_pe_z|\allowbreak\le\alpha\|A\|_2\allowbreak\lambda_{\max}(K_p)\|e_z\|\,\|e_\xi\|$ 并入等效扰动幅值 $D$。

  \item \textbf{$A$ 的谱范数界}：由 $A_{11}^\top A_{11}=\tfrac14(I_3-\mathcal O\mathcal O^\top)$ 得 $\|A_{11}\|_2=\tfrac12$（附录 C.2），由块下三角结构得 $\|A\|_2\le\max\{\|A_{11}\|_2,1\}+\|[\mathcal T]_\times\|_2=1+\|\mathcal T\|$。在 $\Omega_c$ 上 $\|\mathcal T\|\le\sqrt{2c/\lambda_{\min}(K_{p,T})}$、$\|e_z\|\le\sqrt{2c/\lambda_{\min}(K_p)}$，代入第 2 步得 \eqref{eq:5.7b} 与
  \begin{equation}
  \dot V\ \le\ -\lambda_{\mathrm{eff}}\|e_\xi\|^2+D\,\|e_\xi\| .
  \tag{5.7d}\label{eq:5.7d}
  \end{equation}

  \item \textbf{Young 不等式与积分}：$D\|e_\xi\|\le\tfrac{\lambda_{\mathrm{eff}}}2\|e_\xi\|^2+\tfrac{D^2}{2\lambda_{\mathrm{eff}}}$，故 $\dot V\le-\tfrac{\lambda_{\mathrm{eff}}}2\|e_\xi\|^2+\tfrac{D^2}{2\lambda_{\mathrm{eff}}}$。在 $[0,T]$ 上积分并弃去 $V(T)\ge0$：
  \begin{equation*}
  \begin{gathered}
  \tfrac{\lambda_{\mathrm{eff}}}2\int_0^T\|e_\xi\|^2dt
  \ \le\ V(0)+\tfrac{D^2}{2\lambda_{\mathrm{eff}}}\,T ,
  \end{gathered}
  \end{equation*}
  两端除以 $\tfrac{\lambda_{\mathrm{eff}}}2T$ 即 \eqref{eq:5.7c}；令 $T\to\infty$ 得 \eqref{eq:5.7}。$\alpha\to0$ 时 $D\to D_{\mathrm{ex}}$、$\lambda_{\mathrm{eff}}\to\lambda_{\min}(K_d)$，退化为经典形式。
\end{enumerate}

\subsection*{C.9 紧性：认证增益与线性化真实增益相等}

定理 3(c′) 的认证值是否保守，由上、下两个方向夹逼回答。

上界来自证书 \eqref{eq:5.6c}：任意 $L_2$ 扰动，其能量至多按 $1/\lambda_{\min}(K_d)$ 透入误差。

下界来自线性化的频率响应。将闭环 \eqref{eq:5.5} 在 $\tilde x\to1$ 处线性化，两通道方程为 $\ddot{\mathcal O}+K_\omega\dot{\mathcal O}+\tfrac14K_{p,O}\,\mathcal O=-\tfrac12\,d_\omega$ 与 $\ddot{\mathcal T}+K_v\dot{\mathcal T}+k_{p,T}\,\mathcal T=+\,d_v$，其 $d\to e_\xi$ 传递函数均为二阶环节 $G(s)=\dfrac{s}{s^2+bs+c}$：旋转通道由 $\dot e_{z,O}=-\tfrac12\tilde\omega$ 得 $\tilde\omega=-2\dot{\mathcal O}$，化简得 $b=\lambda(K_\omega)$、$c=p_O/4$；平移通道 $\tilde v=\dot{\mathcal T}$，$b=\lambda(K_v)$、$c=k_{p,T}$。对 $G$ 有
\begin{equation}
|G(j\omega)|^2=\frac{u}{(c-u)^2+b^2u},\quad u=\omega^2,\qquad
\frac{\mathrm{d}}{\mathrm{d}u}\,[\,\cdot\,]\propto(c-u)(c+u),
\tag{5.10}\label{eq:5.10}
\end{equation}
故 $|G|$ 于 $u=c$ 取全局最大值 $1/b$，即扰动频率等于通道自然频率 $\sqrt c$ 时（与阻尼大小无关）达到峰值。于是块对角 $K_d$ 下线性化的真实 $H_\infty$ 增益为 $\max\bigl(1/\lambda(K_\omega),\,1/\lambda(K_v)\bigr)=1/\lambda_{\min}(K_d)$。

由非线性闭环的局部 $L_2$ 增益不低于其线性化 $H_\infty$ 范数的标准结论（零初值小信号意义 \cite{VdS00}），结合证书给出的匹配上界，闭环 \eqref{eq:5.5} 的局部 $L_2$ 增益恰为 $1/\lambda_{\min}(K_d)$。附带实验预言：最坏扰动频率为通道自然频率 $\sqrt c$（实机 G4 档即 $6\,\mathrm{rad/s}$）。
